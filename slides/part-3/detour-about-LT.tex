
\begin{frame}{Propriedades sobre Transformações Lineares}

  Sejam $T$, $T_1$ e $T_2$ transformações lineares, então:

  \vspace{0.3cm}

  \begin{itemize}
    \item Distributividade:
      \[
        T(a + b) = T(a) + T(b), \quad \forall a, b \in \mathbb{R}^{n},
      \]

    \item Composição:
      \[
        T_1 \circ T_2 \text{ é uma transformação linear, }
      \]

    \item Split:
      \[
        T(a, b) = T_1(a) + T_2(b), \quad \forall a, b \in \mathbb{R}^{n}.
      \]
  \end{itemize}

\end{frame}
