  \begin{theorem}[Solução de Mínimos Quadrados]
    Sejam $A \in \mathbb{R}^{m \times n}$ uma matriz e $b \in \mathbb{R}^{m}$ um vetor.
    \[
      x^{*} \in \argmin{x} \norm{Ax - b}^2
      \iff
      \exists Q, R \st x^{*} = R^{-1} Q\t b
    \]
  \end{theorem}
