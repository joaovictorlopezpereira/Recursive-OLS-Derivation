
\begin{frame}{Nossa Abordagem (Otimização)}
  \begin{enumerate}
    \item Operar em uma linguagem consistente; \\[0.75cm]
    \item Não realizar trocas de contexto; \\[0.75cm]
    \item Minimizar argumentos de cálculo, geometria e desigualdades. \\[0.75cm]
  \end{enumerate}

  \begin{definition}[Argmin]
    Seja $f : X \mapsto \mathbb{R}$ uma função qualquer.
    \[
      \argmin{x \in X} f(x) = \{x' \in X \st \forall x \in X, f(x') \leq f(x)\}.
    \]
  \end{definition}

\end{frame}
